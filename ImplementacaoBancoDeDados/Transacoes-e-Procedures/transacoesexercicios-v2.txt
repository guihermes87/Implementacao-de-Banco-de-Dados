CREATE or alter PROCEDURE usp_inserirnovousuario

@nome as VARCHAR(50),
@saldoconta as MONEY,
@id as int
AS
BEGIN
BEGIN TRAN
BEGIN TRY
INSERT INTO conta
values(@id,@nome, @saldoconta)
select * from conta;
IF (select count(*) from conta where nome = @nome) >1
BEGIN
ROLLBACK TRAN;
PRINT 'Já existe alguém com o nome' +@nome
END
ELSE
BEGIN
COMMIT TRAN;
PRINT 'Cadastro realizado com sucesso'
END
END TRY
BEGIN CATCH
ROLLBACK TRAN;
PRINT 'DEU ERRO'
PRINT 'NÚMERO:' + CAST(ERROR_NUMBER() AS VARCHAR(10));
PRINT 'MENSAGEM DE ERRO:' + ERROR_MESSAGE();
END CATCH
END

GO

BEGIN TRAN;

    INSERT INTO DEPARTAMENTO (Dnome, Dnumero, Cpf_gerente, Data_inicio_gerente)
    VALUES ('Marketing', 10, (SELECT TOP 1 Cpf FROM FUNCIONARIO), GETDATE());

    SAVE TRAN depto1OK;   -- savepoint após o 1º insert

    INSERT INTO DEPARTAMENTO (Dnome, Dnumero, Cpf_gerente, Data_inicio_gerente)
    VALUES ('Jurídico', 11, (SELECT TOP 1 Cpf FROM FUNCIONARIO), GETDATE());

    -- dentro da transação, os dois aparecem
    SELECT Dnumero, Dnome FROM DEPARTAMENTO WHERE Dnumero IN (10, 11);

    ROLLBACK TRAN depto1OK;   -- desfaz só o 2º insert

COMMIT TRAN;   -- o ROLLBACK para o savepoint não encerra a transação

-- Evidência: após o COMMIT, apenas o departamento 10 (Marketing) existe
SELECT Dnumero, Dnome FROM DEPARTAMENTO WHERE Dnumero IN (10, 11);

DECLARE @cpf      CHAR(11)     = '12345678966';   -- funcionário alvo
DECLARE @novoEnd  VARCHAR(100) = 'Rua das Flores, 100, Santa Maria, RS';
DECLARE @linhas   INT;


SELECT Cpf, Pnome, Endereco AS Endereco_antes FROM FUNCIONARIO WHERE Cpf = @cpf;

BEGIN TRAN;

    UPDATE FUNCIONARIO SET Endereco = @novoEnd WHERE Cpf = @cpf;
    SET @linhas = @@ROWCOUNT;

    -- conferência: alterou exatamente 1 linha e o valor ficou correto?
    IF @linhas = 1
       AND EXISTS (SELECT 1 FROM FUNCIONARIO WHERE Cpf = @cpf AND Endereco = @novoEnd)
    BEGIN
        COMMIT TRAN;
        PRINT 'DECISÃO: COMMIT - 1 registro alterado e conferido.';
    END
    ELSE
    BEGIN
        ROLLBACK TRAN;
        PRINT 'DECISÃO: ROLLBACK - linhas afetadas = ' + CAST(@linhas AS VARCHAR(10));
    END

-- estado final
SELECT Cpf, Pnome, Endereco AS Endereco_final FROM FUNCIONARIO WHERE Cpf = @cpf;


GO
